local function realTP(targetCF, method)
    local char = LocalPlayer.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return end
    method = method or State.RealTPMethod
    if method == "cframe" then
        hrp.CFrame = targetCF
        pcall(function() hrp.AssemblyLinearVelocity = Vector3.zero end)
    elseif method == "velocity" then
        local dir = targetCF.Position - hrp.Position
        if dir.Magnitude < 1 then hrp.CFrame = targetCF
        else pcall(function() hrp.AssemblyLinearVelocity = dir.Unit * math.min(dir.Magnitude * 5, 1000) end) end
    elseif method == "moveto" then
        hum:MoveTo(targetCF.Position)
    end
end
local function tpToPlayer(plr)
    if not plr or not plr.Character then return end
    local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
    if hrp then realTP(hrp.CFrame * CFrame.new(0, 0, -3)) end
end

    bar.Position = UDim2.new(0, 12, 0, 32)
    bar.BackgroundColor3 = Color3.fromRGB(60,60,75)
    bar.BorderSizePixel = 0
    bar.Parent = f
    newCorner(bar, 3)
    local fill = Instance.new("Frame")
    fill.Size = UDim2.new((default-min)/(max-min), 0, 1, 0)
    fill.BackgroundColor3 = Theme.Accent
    fill.BorderSizePixel = 0
    fill.Parent = bar
    newCorner(fill, 3)
    local dragging = false
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(1, 0, 0, 16)
    btn.Position = UDim2.new(0, 0, 0, 26)
    btn.BackgroundTransparency = 1
    btn.Text = ""
    btn.Parent = f
    local function update(input)
        local rel = math.clamp((input.Position.X - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
        local val = min + rel * (max - min)
        if max - min > 20 then val = math.floor(val) end
        fill.Size = UDim2.new(rel, 0, 1, 0)
        lbl.Text = name .. ": " .. val .. (suffix or "")
        if cb then pcall(cb, val) end
    end
    btn.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            update(i)
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then update(i) end
    end)
    UIS.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then dragging = false end
    end)
end

local function stopFly()

local function startFly()
    stopFly()

    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    flyVel = Instance.new("BodyVelocity")

    flyGyro.P = flyConfig.Smoothness * 100
    flyGyro.D = flyConfig.Smoothness * 10

local freecamConn, freecamPos = Vector3.zero, Vector3.zero
local freecamRX, freecamRY = 0, 0
local function setFreecam(enable)
    if enable then
        if State.Freecam then return end
        State.Freecam = true
        Camera.CameraType = Enum.CameraType.Scriptable
        Camera.CameraSubject = nil
        freecamPos = Camera.CFrame.Position
        local look = Camera.CFrame.LookVector
        freecamRY = math.atan2(-look.X, -look.Z)
        freecamRX = math.asin(math.clamp(look.Y, -1, 1))
        if freecamConn then freecamConn:Disconnect() end
        freecamConn = RunService.RenderStepped:Connect(function()
            pcall(function()
                if not State.Freecam then return end
                if UIS:IsMouseButtonPressed(Enum.UserInputType.MouseButton2) then
                    local d = UIS:GetMouseDelta()
                    freecamRY -= d.X * 0.005
                    freecamRX = math.clamp(freecamRX + d.Y * 0.005, -math.pi/2 + 0.01, math.pi/2 - 0.01)
                end
                local speed = UIS:IsKeyDown(Enum.KeyCode.LeftShift) and 3 or 1
                local rot = CFrame.fromEulerAnglesYXZ(freecamRX, freecamRY, 0)
                local dir = Vector3.zero
                if UIS:IsKeyDown(Enum.KeyCode.W) then dir += rot.LookVector end
                if UIS:IsKeyDown(Enum.KeyCode.S) then dir -= rot.LookVector end
                if UIS:IsKeyDown(Enum.KeyCode.A) then dir -= rot.RightVector end
                if UIS:IsKeyDown(Enum.KeyCode.D) then dir += rot.RightVector end
                if UIS:IsKeyDown(Enum.KeyCode.Space) then dir += Vector3.new(0,1,0) end
                if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then dir -= Vector3.new(0,1,0) end
                freecamPos += dir * speed
                Camera.CFrame = CFrame.new(freecamPos) * rot
            end)
        end)
    else
        if not State.Freecam then return end
        State.Freecam = false
        if freecamConn then freecamConn:Disconnect(); freecamConn = nil end
        Camera.CameraType = Enum.CameraType.Custom
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        Camera.CameraSubject = hum or char
    end
end
toggle(MP, "Freecam (F4)", false, function(v) setFreecam(v) end)
button(MP, "回归主视角", function() setFreecam(false) end)

local function log(txt, color)
    local l = Instance.new("TextLabel")
    l.Size = UDim2.new(1, -8, 0, 18)
    l.BackgroundTransparency = 1
    l.Text = "[Xeno] " .. txt
    l.TextColor3 = color or Theme.Green
    l.Font = Theme.FontCode
    l.TextSize = 12
    l.TextXAlignment = Enum.TextXAlignment.Left
    l.Parent = out
    task.wait()
    out.CanvasSize = UDim2.new(0,0,0, ol.AbsoluteContentSize.Y + 8)
end
ci.FocusLost:Connect(function(enter)
    if enter and ci.Text ~= "" then
        log("> "..ci.Text, Color3.fromRGB(255,220,100))
        local cmd, arg = ci.Text:match("^(%S+)%s*(.*)$")
        if cmd then
            cmd = cmd:lower()
            if cmd == "help" then log("tp/speed/god/supergod/noclip/kill/players/clear/lua", Theme.Accent) end
            if cmd == "tp" then local p = Players:FindFirstChild(arg) if p then tpToPlayer(p); log("传送 "..arg) end end
            if cmd == "speed" then local v = tonumber(arg) if v then applySpeed(v); log("速度 "..v) end end
            if cmd == "god" then State.GodMode = not State.GodMode; log("无敌 "..tostring(State.GodMode)) end
            if cmd == "supergod" then State.SuperGodMode = not State.SuperGodMode; log("超无敌 "..tostring(State.SuperGodMode)) end
            if cmd == "noclip" then State.Noclip = not State.Noclip; log("穿墙 "..tostring(State.Noclip)) end
            if cmd == "kill" then local p = Players:FindFirstChild(arg) if p and p.Character then local h = p.Character:FindFirstChildOfClass("Humanoid") if h then h.Health = 0 end log("击杀 "..arg) end end
            if cmd == "players" then for _, p in pairs(Players:GetPlayers()) do log("  "..p.Name) end end
            if cmd == "clear" then for _, c in pairs(out:GetChildren()) do if c:IsA("TextLabel") then c:Destroy() end end end
            if cmd == "lua" then local ok2, err = pcall(loadstring(arg)); log(ok2 and "OK" or ("错误: "..tostring(err)), ok2 and Theme.Green or Theme.Red) end
        end
        ci.Text = ""
    end
end)
