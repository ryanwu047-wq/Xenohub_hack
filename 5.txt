section(A5, "自动农场")
toggle(A5, "46. 农场 (最近玩家)", false, function(v)
    State.AutoFarmNearest = v
    if v then
        task.spawn(function()
            while State.AutoFarmNearest do
                pcall(function()
                    local myHRP = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                    if myHRP then
                        local closest, minD
                        for _, p in pairs(Players:GetPlayers()) do
                            if p == LocalPlayer then continue end
                            local hrp = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
                            local hum = p.Character and p.Character:FindFirstChildOfClass("Humanoid")
                            if hrp and hum and hum.Health > 0 then
                                local d = (hrp.Position - myHRP.Position).Magnitude
                                if not minD or d < minD then minD = d; closest = hrp end
                            end
                        end
                        if closest then realTP(CFrame.new(closest.Position + Vector3.new(0, 3, 0))) end
                    end
                end)
                task.wait(0.5)
            end
        end)
    end
end)
toggle(A5, "47. 农场 (NPC/怪物)", false, function(v)

                                local hum = obj:FindFirstChildOfClass("Humanoid")
                                local hrp = obj:FindFirstChild("HumanoidRootPart")
                                if hum and hrp and hum.Health > 0 and hum.Health < hum.MaxHealth then

                                    break
                                end
                            end
                        end
                    end
                end)
                task.wait(0.5)
            end
        end)
    end
end)
