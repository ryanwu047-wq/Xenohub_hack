toggle(Fun, "51. 彩虹拖尾", false, function(v)
    State.RainbowTrail = v
    if v then
        local hrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if hrp then
            local a1 = Instance.new("Attachment", hrp)
            local a2 = Instance.new("Attachment", hrp)
            a1.Position = Vector3.new(-1, 1, 0)
            a2.Position = Vector3.new(1, -1, 0)
            local t = Instance.new("Trail")
            t.Attachment0 = a1
            t.Attachment1 = a2
            t.Lifetime = 1
            t.Color = ColorSequence.new({
                ColorSequenceKeypoint.new(0, Color3.fromRGB(255,0,0)),
                ColorSequenceKeypoint.new(0.5, Color3.fromRGB(0,255,0)),
                ColorSequenceKeypoint.new(1, Color3.fromRGB(0,0,255)),
            })
            t.Parent = hrp
            State.RainbowTrailObj = t
        end
    else
        if State.RainbowTrailObj then State.RainbowTrailObj:Destroy() end
    end
end)
