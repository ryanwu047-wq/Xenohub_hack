toggle(SpinP, "1. 自转 (相机不动)", false, function(v)
    State.SpinNoCamera = v
    if v then
        task.spawn(function()
            while true do
                pcall(function()
                    if not State.SpinNoCamera then return end
                    local char = LocalPlayer.Character
                    if not char then return end
                    local hrp = char:FindFirstChild("HumanoidRootPart")
                    if not hrp then return end
                    hrp.CFrame = hrp.CFrame * CFrame.Angles(0, math.rad(State.SpinSpeed), 0)
                end)
                RunService.Heartbeat:Wait()
            end
        end)
    end
end)

-- 战斗增强
local CAP = addCategory("战斗增强")

toggle(CAP, "自动开枪 (FOV 内)", false, function(v)
    State.AutoFire = v
    if v then
        task.spawn(function()
            while true do
                pcall(function()
                    if not State.AutoFire then return end
                    local vp = Camera.ViewportSize / 2
                    local target = false
                    for _, p in pairs(Players:GetPlayers()) do
                        if p == LocalPlayer then continue end
                        local hum = p.Character and p.Character:FindFirstChildOfClass("Humanoid")
                        local hrp = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
                        if hum and hrp and hum.Health > 0 then
                            local sp, on = Camera:WorldToViewportPoint(hrp.Position)
                            if on and sp.Z > 0 and (Vector2.new(sp.X, sp.Y) - vp).Magnitude < State.FOV then
                                target = true; break
                            end
                        end
                    end
                    if target then
