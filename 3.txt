-- 掉落物 ESP
local DropP = addCategory("掉落物 ESP")

local dropHighlightEnabled = false
local dropHighlights = {}
task.spawn(function()
    while true do
        pcall(function()
            if dropHighlightEnabled then
                for _, obj in pairs(workspace:GetDescendants()) do
                    if obj:IsA("BasePart") and not dropHighlights[obj] then
                        local n = obj.Name:lower()
                        local isDrop = false
                        if n:find("chest") or n:find("coin") or n:find("gem") or n:find("cash")
                           or n:find("drop") or n:find("loot") or n:find("reward")
                           or n:find("crystal") or n:find("key") or n:find("box") then
                            isDrop = true
                        end
                        if isDrop then
                            local hl = Instance.new("Highlight")
                            hl.Name = "XenoDropHL"
                            hl.FillColor = Color3.fromRGB(255, 200, 0)
                            hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                            hl.FillTransparency = 0.5
                            hl.Adornee = obj
                            hl.Parent = obj
                            dropHighlights[obj] = hl
                        end
                    end
                end
            else
                for obj, hl in pairs(dropHighlights) do
                    if hl and hl.Parent then hl:Destroy() end
                end
                dropHighlights = {}
            end
        end)
        task.wait(2)
    end
end)
toggle(DropP, "掉落物高亮 (箱子/金币/水晶)", false, function(v) dropHighlightEnabled = v end)

local dropTexts = {}
toggle(DropP, "掉落物显示距离", false, function(v)
    State.DropDistance = v
    if v then
        task.spawn(function()
            while true do
                pcall(function()
                    if not State.DropDistance then return end
                    local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                    if not myRoot then return end
                    for obj, hl in pairs(dropHighlights) do
                        if hl and obj.Parent then
                            if not dropTexts[obj] then
                                local bb = Instance.new("BillboardGui")
                                bb.Name = "XenoDropDist"
                                bb.Size = UDim2.new(0, 100, 0, 20)
                                bb.StudsOffsetWorldSpace = Vector3.new(0, 3, 0)
                                bb.AlwaysOnTop = true
                                bb.Parent = obj
                                local lbl = Instance.new("TextLabel", bb)
                                lbl.Size = UDim2.new(1, 0, 1, 0)
                                lbl.BackgroundTransparency = 1
                                lbl.TextColor3 = Color3.fromRGB(255, 200, 0)
                                lbl.TextStrokeTransparency = 0
                                lbl.Font = Enum.Font.GothamBold
                                lbl.TextSize = 12
                                dropTexts[obj] = lbl
                            end
                            local dist = (obj.Position - myRoot.Position).Magnitude
                            dropTexts[obj].Text = math.floor(dist) .. " studs"
                        end
                    end
                    for obj, lbl in pairs(dropTexts) do
                        if not obj.Parent then
                            if lbl.Parent then lbl.Parent:Destroy() end
                            dropTexts[obj] = nil
                        end
                    end
                end)
                task.wait(0.2)
            end
        end)
    end
end)

-- 玩家雷达
local RadarP = addCategory("玩家雷达")

local radarEnabled = false
local radarFrame = Instance.new("Frame")
radarFrame.Size = UDim2.new(0, 200, 0, 200)
radarFrame.Position = UDim2.new(1, -220, 0, 100)
radarFrame.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
radarFrame.BackgroundTransparency = 0.5
radarFrame.BorderSizePixel = 1
radarFrame.BorderColor3 = Color3.fromRGB(90, 150, 255)
radarFrame.Visible = false
radarFrame.Parent = ScreenGui
Instance.new("UICorner", radarFrame).CornerRadius = UDim.new(0, 8)

local radarDots = {}
task.spawn(function()
    while true do
        pcall(function()
            if radarEnabled then
                radarFrame.Visible = true
                local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                if not myRoot then return end
                local myPos = myRoot.Position
                local myAngle = math.atan2(-Camera.CFrame.LookVector.X, -Camera.CFrame.LookVector.Z)
                for _, p in pairs(Players:GetPlayers()) do
                    if p == LocalPlayer then continue end
                    local hrp = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
                    if not hrp then
                        if radarDots[p] then radarDots[p]:Destroy(); radarDots[p] = nil end
                        continue
                    end
                    local rel = hrp.Position - myPos
                    local dist = math.sqrt(rel.X^2 + rel.Z^2)
                    if dist > 300 then
                        if radarDots[p] then radarDots[p].Visible = false end
                        continue
                    end
                    local angle = math.atan2(rel.X, rel.Z) - myAngle
                    local x = math.sin(angle) * (dist / 300) * 90 + 100
                    local y = math.cos(angle) * (dist / 300) * 90 + 100
                    if not radarDots[p] then
                        local dot = Instance.new("Frame")
                        dot.Size = UDim2.new(0, 6, 0, 6)
                        dot.BackgroundColor3 = Color3.fromRGB(255, 0, 0)
                        dot.BorderSizePixel = 0
                        dot.Parent = radarFrame
                        Instance.new("UICorner", dot).CornerRadius = UDim.new(0, 3)
                        radarDots[p] = dot
                    end
                    radarDots[p].Position = UDim2.new(0, x - 3, 0, y - 3)
                    radarDots[p].Visible = true
                end
            else
                radarFrame.Visible = false
                for p, dot in pairs(radarDots) do
                    dot:Destroy()
                end
                radarDots = {}
            end
        end)
        task.wait(0.1)
    end
end)
toggle(RadarP, "玩家雷达 (200x200)", false, function(v) radarEnabled = v end)
