                pcall(function()
                    local char = LocalPlayer.Character
                    local backpack = LocalPlayer:FindFirstChildOfClass("Backpack")
                    if char and backpack then
                        for _, tool in pairs(backpack:GetChildren()) do
                            if tool:IsA("Tool") then

                        end
                    end
                end)
                task.wait(1)
            end
        end)
    end
